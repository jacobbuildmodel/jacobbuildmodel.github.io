{{- /* One "fork" panel of the LTA road figure: a single lane for all cars splitting into bands. Band order only (EDGE_C 3.2 option b): LTA publishes no OMV by model, so there is no value axis and no claim about where a model sits inside its band. Called by lta_road.html with a dict: id unique prefix for title/desc ids n number of bands (3 or 5) h, s lane height and lane pitch, in viewBox units amts signed amount per band, as display text forks LTA's percentile at each fork, as display text (n-1 items) dyn 0-based band of the BYD Sealion 7 Dynamic perf 0-based band of the BYD Sealion 7 Performance gap the two trims' difference, as display text The viewBox is 360 wide, close to the 362 px content column at a 390 px phone, so 12 px type renders at about 12 px. */}} {{- $n := .n }} {{- $h := .h }} {{- $s := .s }} {{- $top := 8.0 }} {{- $W := 360.0 }} {{- $cy := slice }} {{- range $i := seq 0 (sub $n 1) }} {{- $cy = $cy | append (add (add $top (div $h 2.0)) (mul $i $s)) }} {{- end }} {{- $first := index $cy 0 }} {{- $last := index $cy (sub $n 1) }} {{- $ym := div (add $first $last) 2.0 }} {{- $H := add (add $last (div $h 2.0)) $top }} {{- $x1 := 58.0 }}{{ $x2 := 126.0 }}{{ $tx := 134.0 }}{{ $dx := 240.0 }} {{- $paths := slice }} {{- range $cy }} {{- $paths = $paths | append (printf "M0,%.1f L%.1f,%.1f C%.1f,%.1f %.1f,%.1f %.1f,%.1f L%.1f,%.1f" $ym $x1 $ym (add $x1 34.0) $ym (sub $x2 34.0) . $x2 . $W .) }} {{- end }} {{- $words := dict "3" "three" "5" "five" }} {{- $word := index $words (string $n) }} {{- $forkList := delimit .forks ", " " and " }}
